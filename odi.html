<script>

    function calculateODI() {

        var totalScore = 0;

        var questionsAnswered = 0;


        for (var i = 1; i <= 10; i++) {

            var radios = document.getElementsByName('q' + i);

            for (var j = 0; j < radios.length; j++) {

                if (radios[j].checked) {

                    totalScore += parseInt(radios[j].value);

                    questionsAnswered++;

                }

            }

        }


        if (questionsAnswered === 0) {

            alert("Please answer the questions first. / الرجاء الإجابة على الأسئلة أولاً");

            return;

        }


        var maxPossibleScore = questionsAnswered * 5;

        var percentage = (totalScore / maxPossibleScore) * 100;

        percentage = Math.round(percentage);


        var category = "";

        var arCategory = "";

        

        if (percentage <= 20) { category = "Minimal Disability"; arCategory = "إعاقة طفيفة"; }

        else if (percentage <= 40) { category = "Moderate Disability"; arCategory = "إعاقة متوسطة"; }

        else if (percentage <= 60) { category = "Severe Disability"; arCategory = "إعاقة شديدة"; }

        else if (percentage <= 80) { category = "Crippled"; arCategory = "إعاقة مقعدة"; }

        else { category = "Bed-bound"; arCategory = "طريح الفراش"; }


        document.getElementById('score-display').innerText = percentage + "%";

        document.getElementById('category-display').innerText = category + " - " + arCategory;
